agot_burn_your_enemies_dragon_raid_capital_interaction = {
    category = interaction_category_hostile
    desc = agot_burn_your_enemies_dragon_raid_capital_interaction_desc
    icon = icon_dragon_burn

    # Only allows to if both not at war or not imprisoned
    is_shown = {
        scope:actor = {
            OR = {
                has_trait = dragonrider
            }
            is_imprisoned = no
            is_at_war = no
            is_travelling = no
        }
        scope:recipient = {
            is_imprisoned = no
            is_landed = yes
            is_at_war = no
            is_travelling = no
            NOT = { is_imprisoned = yes }
            exists = primary_title
            exists = capital_county
            NOT = { this = scope:actor }
        }
    }

    # Cooldown check
    is_valid = {
        scope:actor = {
            NOT = { has_character_flag = agot_burn_your_enemies_dragon_raid_cooldown }
        }
    }

	auto_accept = yes
    use_diplomatic_range = yes

    on_accept = {
        # Apply cooldown
        scope:actor = {
            add_character_flag = {
                name = agot_burn_your_enemies_dragon_raid_cooldown
                years = 10
            }
            # Save the player's dragon using AGOT's native variable
            var:current_dragon = {
                save_scope_as = agot_burn_your_enemies_my_dragon
            }
        }
        
        # Save target capital county and capital barony correctly
        scope:recipient = {
            capital_county = {
                save_scope_as = agot_burn_your_enemies_target_capital_county
            }
            # Use capital_holding directly from the character to avoid tier errors
            capital_holding = {
                save_scope_as = agot_burn_your_enemies_target_capital_barony
            }
        }

        # Trigger event chain using explicit bracket syntax
        scope:actor = {
            trigger_event = {
                id = agot_burn_your_enemies_dragon_raid.0001
            }
        }
    }

    ai_will_do = {
        base = 0 # Keeps AI from spamming it on players
    }
}
