agot_burn_your_enemies_minor_scorpion_damage_effect = {
	if = {
		limit = {
			scope:agot_burn_your_enemies_target_capital_barony = {
				has_stationed_scorpions = yes
			}
			agot_burn_your_enemies_dragon_attack_score < agot_burn_your_enemies_dragon_defense_score
		}
		random_list = {
			70 = {}
			25 = {
				scope:actor = {
					add_trait = wounded_1
				}
				scope:agot_burn_your_enemies_my_dragon = {
					wound_dragon = yes
					add_character_flag = {
						flag = agot_burn_your_enemies_dragon_shaken
						years = 3
					}
				}
				# NEW: show the dragon taking the hit
				scope:actor = {
					trigger_event = {
						id = agot_burn_your_enemies_dragon_raid.0005
					}
				}
			}
			5 = {
				# Rare but possible: heavily outmatched by the defenses, the
				# scorpions land a killing blow on the dragon itself.
				trigger = {
					agot_burn_your_enemies_dragon_defense_margin > 30
				}
				scope:agot_burn_your_enemies_my_dragon = {
					death = {
						death_reason = death_agot_burn_your_enemies_scorpion_kill
					}
				}
				scope:actor = {
					trigger_event = {
						id = agot_burn_your_enemies_dragon_raid.0008
					}
				}
			}
		}
	}
}
