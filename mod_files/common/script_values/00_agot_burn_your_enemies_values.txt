# Uses AGOT's native dragon_combat_effectiveness stat
agot_burn_your_enemies_dragon_attack_score = {
    value = scope:actor.prowess
    if = {
        limit = { exists = scope:agot_burn_your_enemies_my_dragon }
        add = scope:agot_burn_your_enemies_my_dragon.dragon_combat_effectiveness
    }
}

agot_burn_your_enemies_dragon_defense_score = {
    value = 0
    if = {
        limit = { exists = scope:agot_burn_your_enemies_target_capital_barony }
        value = scope:agot_burn_your_enemies_target_capital_barony.fort_level
        multiply = 5
    }
}
