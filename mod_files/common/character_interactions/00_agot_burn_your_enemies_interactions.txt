agot_burn_your_enemies_dragon_raid_capital_interaction = {
    category = interaction_category_hostile
    desc = agot_burn_your_enemies_dragon_raid_capital_interaction_desc
    icon = faction

    is_shown = {
        scope:actor = {
            AND = {
                has_trait = dragonrider
                has_trait = vengeful
            }
        }
        scope:recipient = {
            is_landed = yes
            exists = primary_title
            exists = capital_county
        }
    }

    is_valid_showing_failures_only = {
        scope:actor = {
            custom_tooltip = {
                text = agot_burn_your_enemies_actor_vengeful_trait
                has_trait = vengeful
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_actor_at_war
                is_at_war = no
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_actor_is_imprisoned
                is_imprisoned = no
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_actor_travelling
                is_travelling = no
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_actor_cooldown
                NOT = { has_character_flag = agot_burn_your_enemies_actor_cooldown }
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_actor_no_dragon
                has_variable = current_dragon
            }
        }
        scope:recipient = {
            custom_tooltip = {
                text = agot_burn_your_enemies_recipient_at_war
                is_at_war = no
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_recipient_travelling
                is_travelling = no
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_receipient_is_imprisoned
                is_imprisoned = no
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_not_allied
                NOT = { is_allied_to = scope:actor }
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_no_truce
                NOT = { has_truce = scope:actor }
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_not_in_your_realm
                NOT = { is_vassal_or_below_of = scope:actor }
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_capital_already_ruined
                NOT = {
                    capital_province = {
                        has_holding_type = ruin_holding
                    }
                }
            }
            custom_tooltip = {
                has_relation_rival = scope:actor
                text = agot_burn_your_enemies_not_your_rival
            }
            custom_tooltip = {
                text = agot_burn_your_enemies_no_court_dragonrider
                NOT = {
                    any_courtier_or_guest = {
                        is_travelling = no
                        OR = {
                            has_trait = dragonrider
                            has_trait = dragon
                        }
                    }
                }
            }
        }
    }

	auto_accept = yes
    use_diplomatic_range = yes

    on_accept = {
        scope:actor = {
            add_character_flag = {
                flag = agot_burn_your_enemies_actor_cooldown
                years = 20
            }
            if = {
                limit = { exists = var:current_dragon }
                set_variable = {
                    name = agot_burn_your_enemies_my_dragon_var
                    value = var:current_dragon
                }
            }
            set_variable = {
                name = agot_burn_your_enemies_target_var
                value = scope:recipient
            }
        }

        scope:recipient = {
            capital_county = {
                save_scope_as = agot_burn_your_enemies_target_capital_county
            }
            capital_province = {
                save_scope_as = agot_burn_your_enemies_target_capital_barony
            }
        }

        scope:actor = {
            set_variable = {
                name = agot_burn_your_enemies_target_capital_county_var
                value = scope:agot_burn_your_enemies_target_capital_county
            }
            set_variable = {
                name = agot_burn_your_enemies_target_capital_barony_var
                value = scope:agot_burn_your_enemies_target_capital_barony
            }
        }

        hidden_effect = {
            scope:actor = {
                start_travel_plan = {
                    destination = scope:agot_burn_your_enemies_target_capital_barony
                    players_use_planner = no
                    on_arrival_event = agot_burn_your_enemies_dragon_raid.0001
                    on_arrival_destinations = all_but_last
                    save_scope_as = agot_burn_your_enemies_travel_plan
                }
            }

            scope:agot_burn_your_enemies_travel_plan = {
                add_travel_option = dragon_flight_option
            }
        }
    }


    ai_will_do = {
        base = 0
    }
}
