namespace = agot_burn_your_enemies_dragon_raid

agot_burn_your_enemies_dragon_raid.0001 = {
    type = character_event
    window = agot_dragon_character_event
    title = agot_burn_your_enemies_dragon_raid.0001.t
    desc = agot_burn_your_enemies_dragon_raid.0001.desc
    theme = dragon

    override_background = {
        reference = burning_building
    }

    immediate = {
        scope:actor = {
            var:agot_burn_your_enemies_target_var = {
                save_scope_as = agot_burn_your_enemies_target
            }
            var:agot_burn_your_enemies_my_dragon_var = {
                save_scope_as = agot_burn_your_enemies_my_dragon
            }
            var:agot_burn_your_enemies_target_capital_county_var = {
                save_scope_as = agot_burn_your_enemies_target_capital_county
            }
            var:agot_burn_your_enemies_target_capital_barony_var = {
                save_scope_as = agot_burn_your_enemies_target_capital_barony
            }
        }

        scope:agot_burn_your_enemies_target = {
            save_scope_as = recipient
        }

        agot_set_dragon_animation_flag_effect = { FLAG = dragon_flying RIDING_DRAGON = yes }
    }

    left_portrait = {
        character = scope:agot_burn_your_enemies_my_dragon
    }

    right_portrait = {
        character = root
        camera = camera_dragonrider_flying_event_front
        animation = dragon_main
        outfit_tags = { dragonrider }
    }

    option = {
        name = agot_burn_your_enemies_dragon_raid.0001.proceed
        trigger = {
            agot_burn_your_enemies_raid_still_valid_trigger = yes
        }

        if = {
            limit = {
                scope:agot_burn_your_enemies_target_capital_barony = {
                    has_stationed_scorpions = yes
                }
                agot_burn_your_enemies_dragon_attack_score < agot_burn_your_enemies_dragon_defense_score
            }
            trigger_event = {
                id = agot_burn_your_enemies_dragon_raid.0004
            }
        }
        else = {
            trigger_event = {
                id = agot_burn_your_enemies_dragon_raid.0002
            }
        }
    }

    option = {
        name = agot_burn_your_enemies_dragon_raid.0001.aborted
        trigger = {
            NOT = { agot_burn_your_enemies_raid_still_valid_trigger = yes }
        }

        add_dread = -10
        add_stress = 10

        scope:actor = {
            agot_burn_your_enemies_clear_raid_variables_effect = yes
        }
    }

    after = {
        agot_clear_dragon_animation_flags = yes
    }
}

# Main Raid Event
agot_burn_your_enemies_dragon_raid.0002 = {
    type = character_event
    window = agot_dragon_character_event
    title = agot_burn_your_enemies_dragon_raid.0002.t
    desc = agot_burn_your_enemies_dragon_raid.0002.desc
    theme = dragon
    override_background = {
        reference = burning_building
    }

    # AGOT Fire & Spark Shaders
    widgets = {
        widget = {
            gui = "agot_event_shader_sparks"
            container = "agot_shader_event_foreground"
        }
        widget = {
            gui = "agot_event_shader_fire"
            container = "agot_shader_event_background"
        }
    }

    left_portrait = {
        character = scope:agot_burn_your_enemies_my_dragon
    }

    right_portrait = {
        character = root
        camera = camera_dragonrider_flying_event_front
        animation = dragon_main
        outfit_tags = { dragonrider }
    }

    immediate = {
        if = {
            limit = { NOT = { exists = scope:agot_burn_your_enemies_my_dragon } }
            var:current_dragon = { save_scope_as = agot_burn_your_enemies_my_dragon }
        }

        if = {
            limit = { NOT = { exists = scope:agot_burn_your_enemies_target_capital_barony } }
            scope:recipient = {
                save_scope_as = agot_burn_your_enemies_target_capital_county
                capital_province = {
                    save_scope_as = agot_burn_your_enemies_target_capital_barony
                }
            }
        }

        agot_set_dragon_animation_flag_effect = { FLAG = dragon_flying RIDING_DRAGON = yes }
    }

    option = {
        name = agot_burn_your_enemies_dragon_raid.0002.minor

        ai_chance = {
            base = 40
            modifier = {
                factor = 0.4
                scope:actor = { has_trait = wrathful }
            }
            modifier = {
                factor = 0.4
                scope:actor = { has_trait = callous }
            }
            modifier = {
                factor = 1.5
                scope:actor = { has_trait = compassionate }
            }
            modifier = {
                factor = 1.5
                scope:actor = { has_trait = content }
            }
        }
        scope:agot_burn_your_enemies_target_capital_county = {
            add_county_modifier = {
                modifier = agot_burn_your_enemies_holding_dragon_scathed
                years = 5
            }
        }
        agot_burn_your_enemies_minor_scorpion_damage_effect = yes

        scope:recipient = {
            add_trait = wounded_1
            add_trait = burned
            add_trait_xp = {
                trait = burned
                value = 30
            }
            add_opinion = {
                target = scope:actor
                modifier = agot_burn_your_enemies_dynasty_feud
            }
            if = {
                limit = {
                    can_set_relation_rival_if_adult_trigger = { CHARACTER = scope:actor }
                }
                set_relation_rival = {
                    target = scope:actor
                    reason = rival_burned_my_capital
                }
            }
        }

        scope:actor = {
            add_trait = wounded_1
            add_dread = 25
            add_stress = 25
            add_prestige = -500
            add_piety = -500
        }

        trigger_event = {
            id = agot_burn_your_enemies_dragon_raid.0003
        }
    }

    option = {
        name = agot_burn_your_enemies_dragon_raid.0002.major

        ai_chance = {
            base = 30
            modifier = {
                factor = 1.6
                scope:actor = { has_trait = wrathful }
            }
            modifier = {
                factor = 1.6
                scope:actor = { has_trait = callous }
            }
            modifier = {
                factor = 1.3
                scope:actor = { has_trait = arrogant }
            }
            modifier = {
                factor = 0.5
                scope:actor = { has_trait = compassionate }
            }
        }

        scope:agot_burn_your_enemies_target_capital_county = {
            change_development_level = -50
            add_county_modifier = {
                modifier = agot_burn_your_enemies_dragon_burned_devastation
                years = 10
            }
        }
        agot_burn_your_enemies_major_scorpion_damage_effect = yes

        scope:recipient = {
            add_trait = wounded_2
            add_trait = burned
            add_trait_xp = {
                trait = burned
                value = 50
            }
            add_opinion = {
                target = scope:actor
                modifier = agot_burn_your_enemies_dynasty_feud
            }
            if = {
                limit = {
                    can_set_relation_rival_if_adult_trigger = { CHARACTER = scope:actor }
                }
                set_relation_rival = {
                    target = scope:actor
                    reason = rival_burned_my_capital
                }
            }
            every_courtier_or_guest = {
                limit = {
                    is_alive = yes
                }
                random_list = {
                    50 = {
                        add_trait = wounded_2
                        add_trait = burned
                        add_trait_xp = {
                            trait = burned
                            value = 50
                        }
                    }
                    50 = {
                        add_trait = wounded_1
                    }
                }
            }
        }

        scope:actor = {
            random_list = {
                40 = {
                    add_trait = wounded_2
                    add_trait = burned
                    add_trait_xp = {
                        trait = burned
                        value = 50
                    }
                    add_dread = 50
                    add_stress = 50
                }
                60 = {
                    add_trait = wounded_1
                    add_dread = 25
                    add_stress = 25
                }
            }
            add_prestige = -10000
            add_piety = -1000
        }

        trigger_event = {
            id = agot_burn_your_enemies_dragon_raid.0003
        }
    }

    option = {
        name = agot_burn_your_enemies_dragon_raid.0002.ruin

        trigger = {
            scope:agot_burn_your_enemies_my_dragon = {
                dragon_combat_effectiveness > 200
            }
        }

        ai_chance = {
            base = 10
            modifier = {
                factor = 2
                scope:actor = {
                    AND = {
                        has_trait = wrathful
                        has_trait = callous
                    }
                }
            }
            modifier = {
                factor = 1.5
                scope:actor = { dread > 70 }
            }
            modifier = {
                factor = 0.3
                scope:actor = { has_trait = compassionate }
            }
            modifier = {
                factor = 0.3
                scope:actor = { has_trait = forgiving }
            }
        }

        scope:agot_burn_your_enemies_target_capital_county = {
            change_development_level = -100
            add_county_modifier = {
                modifier = agot_burn_your_enemies_dragon_burned_devastation
                years = 10
            }
        }
        scope:agot_burn_your_enemies_target_capital_barony = {
            set_holding_type = ruin_holding
        }

        play_music_cue = CK3AGOT_dragon_burn_holding
        agot_burn_your_enemies_major_scorpion_damage_effect = yes

        scope:recipient = {
            every_courtier_or_guest = {
                limit = {
                    is_alive = yes
                }
                death = {
                    death_reason = death_dragon_fire
                    killer = scope:agot_burn_your_enemies_my_dragon
                }
            }
            if = {
                limit = {
                    is_alive = yes
                }
                death = {
                    death_reason = death_dragon_fire
                    killer = scope:agot_burn_your_enemies_my_dragon
                }
            }
        }

        scope:actor = {
            random_list = {
                20 = {
                    death = {
                        death_reason = death_burning_enemy_with_dragon
                    }
                }
                80 = {
                    add_trait = wounded_2
                    add_dread = 100
                    add_stress = 100
                    add_prestige = -10000
                    add_piety = -1000
                }
            }
        }

        if = {
            limit = {
                scope:actor = {
                    is_alive = yes
                }
            }
            trigger_event = {
                id = agot_burn_your_enemies_dragon_raid.0003
            }
        }

        add_internal_flag = dangerous
    }

    option = {
        name = agot_burn_your_enemies_dragon_raid.0002.cancel

        ai_chance = {
            base = 15
            modifier = {
                factor = 3
                scope:actor = { has_trait = compassionate }
            }
            modifier = {
                factor = 2
                scope:actor = { has_trait = forgiving }
            }
            modifier = {
                factor = 0.3
                scope:actor = { has_trait = wrathful }
            }
            modifier = {
                factor = 0.3
                scope:actor = { has_trait = callous }
            }
        }

        if = {
            limit = { has_trait = craven }
        }
        else = {
            add_trait = craven
        }

        random_list = {
            50 = {
                add_opinion = {
                    target = scope:recipient
                    modifier = agot_burn_your_enemies_tried_to_kill_opinion
                }
            }
            20 = {
                add_stress = 100
                add_dread = -50
            }
            10 = {
                agot_burn_your_enemies_minor_scorpion_damage_effect
            }
            10 = {
                agot_burn_your_enemies_major_scorpion_damage_effect
            }
        }
    }

    after = {
        agot_clear_dragon_animation_flags = yes
    }
}
